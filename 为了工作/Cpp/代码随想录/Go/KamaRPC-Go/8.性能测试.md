# 8. 性能测试

做这种基建类项目，性能测试是非常重要的一环。RPC 框架本身处在系统通信的核心位置，它**的吞吐能力、延迟表现以及在高并发场景下的稳定性，都会直接影响到整个系统的性能表现**。因此，在开发过程中，需要通过系统化的压测方式来评估框架的性能指标，例如 QPS、平均延迟以及延迟分布等。

不过要注意一个点(**本项目性能测试 是在我本人的电脑上测试的,根据每个人的电脑配置差异,测出来的性能结果也会有所不一致**)



为了更全面地评估系统性能，我这里实现了两种不同的性能测试方式。一种是基于固定请求数量的性能测试，另一种是基于固定时间窗口的性能测试。这两种方式在测试逻辑、统计方式以及适用场景上都有明显差异，因此分别适用于不同的性能评估目标。

# 第一种测试
第一种方式是**基于固定请求数量的性能测试**。这种测试方式会预先设定一个总请求数量，然后在指定并发数的情况下发送这些请求，直到所有请求都完成后再统计整体性能指标。在这种模式下，测试流程通常是先记录测试开始时间，然后启动多个并发客户端（通过 goroutine 实现），每个客户端负责发送一部分请求。所有请求发送完成并收到响应之后，程序记录结束时间，通过总请求数除以总耗时来计算系统的整体吞吐能力（QPS）。同时，每个请求在发送之前会记录开始时间，在收到响应之后记录结束时间，从而统计单个请求的延迟数据，并进一步计算平均延迟或延迟分布。

我的实现中，这种模式主要用于测试异步接口。客户端负责自行维护请求发送和响应接收的策略，通过 future 或回调的方式处理返回结果。这种设计可以让客户端在发送请求后继续发送新的请求，而不是阻塞等待返回，从而更好地测试 RPC 框架在高并发场景下的处理能力。

这种固定请求数的压测方式具有比较明显的优点。由于请求总数是固定的，每次测试都会执行完全相同数量的请求，**因此测试结果具有较好的可重复性**。在进行性能优化或不同实现方式对比时，这种方式非常有价值。例如在比较不同序列化协议（如 JSON 与 Protobuf）、不同编解码实现或不同连接模型时，通过保持请求数量一致，可以更清晰地观察优化前后的性能差异。

不过，这种模式也存在一定的局限性。由于统计结果需要等待所有请求都完成之后才能计算，如果某些请求出现较高延迟，那么整个测试时间会被这些尾部请求拉长。另外，这种模式在某些实现中可能更像是“批量执行一组请求”，而不是持续稳定地产生负载，因此在模拟真实线上流量方面会存在一定偏差。

```go
package main

import (
	"context"
	"flag"
	"fmt"
	"kamaRPC/internal/client"
	"kamaRPC/internal/codec"
	"kamaRPC/internal/registry"
	"kamaRPC/internal/transport"
	"kamaRPC/pkg/api"
	"log"
	"sync"
	"sync/atomic"
	"time"
)

var (
	concurrency = flag.Int("c", 100, "并发客户端数量")
	total       = flag.Int("n", 10000, "请求总数")
	batchSize   = flag.Int("b", 100, "异步请求的批处理大小")
	etcdAddr    = flag.String("etcd", "localhost:2379", "etcd的地址")
	serviceName = flag.String("s", "Arith", "服务名")
	methodName  = flag.String("m", "Add", "方法名")
)

type callResult struct {
	future   *transport.Future
	reqStart time.Time
}

func main() {
	flag.Parse()

	if *concurrency < 1 {
		*concurrency = 1
	}
	if *batchSize < 1 {
		*batchSize = 1
	}

	log.Printf("Starting benchmark: concurrency=%d, total=%d, batch=%d\n", *concurrency, *total, *batchSize)

	reg, err := registry.NewRegistry([]string{*etcdAddr})
	if err != nil {
		log.Fatalf("Failed to create registry: %v", err)
	}
	defer reg.Close()

	c, err := client.NewClient(
		reg,
		client.WithClientCodec(codec.JSON),
	)
	if err != nil {
		log.Fatalf("Failed to create client: %v", err)
	}
	defer c.Close()

	var (
		wg           sync.WaitGroup
		successCount int64
		failCount    int64
		totalLatency int64
	)

	start := time.Now()

	base := *total / *concurrency
	rem := *total % *concurrency
	// 创建 concurrency 个并发客户端
	for i := 0; i < *concurrency; i++ {
		wg.Add(1)
		reqs := base
		if i < rem {
			reqs++
		}

		go func(requestsPerClient int) {
			defer wg.Done()
			//每一个客户端都有 requestsPerClient 个请求
			remaining := requestsPerClient
			for remaining > 0 {
				//一批一批地发数据
				currentBatch := *batchSize
				if remaining < currentBatch {
					currentBatch = remaining
				}

				var calls []callResult

				for j := 0; j < currentBatch; j++ {
					args := &api.Args{A: j, B: j}
					reqStart := time.Now()

					f, err := c.InvokeAsync(context.Background(), *serviceName, *methodName, args)
					if err != nil {
						atomic.AddInt64(&failCount, 1)
						continue
					}
					calls = append(calls, callResult{future: f, reqStart: reqStart})
				}

				doneCh := make(chan callResult, len(calls))
				for _, item := range calls {
					go func(it callResult) {
						<-it.future.DoneChan()
						doneCh <- it
					}(item)
				}

				for k := 0; k < len(calls); k++ {
					item := <-doneCh
					reply := &api.Reply{}
					if err := item.future.GetResult(reply); err != nil {
						atomic.AddInt64(&failCount, 1)
					} else {
						atomic.AddInt64(&successCount, 1)
						latency := time.Since(item.reqStart).Microseconds()
						atomic.AddInt64(&totalLatency, latency)
					}
				}

				remaining -= currentBatch
			}
		}(reqs)
	}

	wg.Wait()
	duration := time.Since(start)

	qps := float64(successCount) / duration.Seconds()
	var avgLatency float64
	if successCount > 0 {
		avgLatency = float64(totalLatency) / float64(successCount) / 1000.0
	}

	fmt.Println("\nBenchmark Result:")
	fmt.Printf("Total Requests: %d\n", *total)
	fmt.Printf("Concurrency:    %d\n", *concurrency)
	fmt.Printf("Batch Size:     %d\n", *batchSize)
	fmt.Printf("Duration:       %v\n", duration)
	fmt.Printf("Success:        %d\n", successCount)
	fmt.Printf("Failed:         %d\n", failCount)
	fmt.Printf("QPS:            %.2f\n", qps)
	fmt.Printf("Avg Latency:    %.2f ms\n", avgLatency)
}

```

# 第二种测试


第二种方式是基于固定时间窗口的性能测试。**在这种模式下，不再预先设定请求数量，而是设定一个测试持续时间，例如运行 10 秒或 30 秒(可以由命令行选项指定)**。在测试开始之后，程序会启动多个并发客户端，每个客户端在整个测试时间内持续不断地发送 同步RPC(**该场景用同步好一些**) 请求。当达到设定的时间上限之后，客户端停止发送新的请求，然后统计在这段时间内系统总共处理了多少请求。

在这种模式下，QPS 的计算方式是通过“成功请求数量除以测试持续时间”得到。同时，在每个请求发送之前记录开始时间，在收到响应之后记录结束时间，用于统计延迟数据。通过收集这些延迟样本，可以进一步计算平均延迟以及多个百分位指标，例如 P50、P90 和 P99 等。

这些百分位指标在系统性能评估中非常重要。例如 P50 表示 50% 的请求延迟小于该值，P90 表示 90% 的请求延迟小于该值，而 P99 表示 99% 的请求延迟小于该值。相比单纯的平均延迟，这些指标能够更真实地反映系统的延迟分布情况，**尤其是尾延迟**。在很多在线系统中，少量高延迟请求往往会显著影响用户体验，因此尾延迟指标是性能测试中的关键数据。

基于时间窗口的压测方式通常能够保持持续稳定的负载，因为客户端在整个测试周期内都会不断发送请求，使系统始终处于压力状态。这种方式更接近真实生产环境中的请求模式，因为在线系统通常是持续接收请求流量，而不是一次性处理固定数量的请求。因此，这种模式更适合用于评估系统在稳定压力下的整体吞吐能力和延迟表现。

```go
package main

import (
	"context"
	"flag"
	"fmt"
	"kamaRPC/internal/client"
	"kamaRPC/internal/codec"
	"kamaRPC/internal/registry"
	"kamaRPC/pkg/api"
	"log"
	"sort"
	"sync"
	"sync/atomic"
	"time"
)

var (
	concurrency = flag.Int("c", 50, "并发客户端数量")
	durationSec = flag.Int("d", 10, "基准测试持续时间（单位为秒）")
	etcdAddr    = flag.String("etcd", "localhost:2379", "etcd的地址")
	serviceName = flag.String("s", "Arith", "服务名")
	methodName  = flag.String("m", "Add", "方法名")
)

type metrics struct {
	success int64
	fail    int64
	latency []int64
	mu      sync.Mutex
}

func (m *metrics) recordLatency(us int64) {
	m.mu.Lock()
	m.latency = append(m.latency, us)
	m.mu.Unlock()
}

func main() {

	flag.Parse()

	log.Printf("Starting benchmark: concurrency=%d duration=%ds\n",
		*concurrency, *durationSec)

	reg, err := registry.NewRegistry([]string{*etcdAddr})
	if err != nil {
		log.Fatal(err)
	}
	defer reg.Close()

	c, err := client.NewClient(
		reg,
		client.WithClientCodec(codec.JSON),
	)
	if err != nil {
		log.Fatal(err)
	}
	defer c.Close()

	var wg sync.WaitGroup
	var m metrics

	ctx, cancel := context.WithTimeout(
		context.Background(),
		time.Duration(*durationSec)*time.Second,
	)
	defer cancel()

	start := time.Now()

	for i := 0; i < *concurrency; i++ {

		wg.Add(1)

		go func() {

			defer wg.Done()

			for {

				select {

				case <-ctx.Done():
					return

				default:

					args := &api.Args{A: 1, B: 2}

					reqStart := time.Now()

					reply := &api.Reply{}

					err := c.Invoke(
						context.Background(),
						*serviceName,
						*methodName,
						args,
						reply,
					)

					lat := time.Since(reqStart).Microseconds()

					if err != nil {
						atomic.AddInt64(&m.fail, 1)
						continue
					}

					atomic.AddInt64(&m.success, 1)

					m.recordLatency(lat)
				}
			}

		}()
	}

	wg.Wait()

	totalTime := time.Since(start)

	printStats(&m, totalTime)
}

func printStats(m *metrics, duration time.Duration) {

	total := m.success + m.fail

	qps := float64(m.success) / duration.Seconds()

	sort.Slice(m.latency, func(i, j int) bool {
		return m.latency[i] < m.latency[j]
	})

	p50 := percentile(m.latency, 50)
	p90 := percentile(m.latency, 90)
	p99 := percentile(m.latency, 99)

	var avg float64

	if len(m.latency) > 0 {

		var sum int64

		for _, v := range m.latency {
			sum += v
		}

		avg = float64(sum) / float64(len(m.latency)) / 1000
	}

	fmt.Println()
	fmt.Println("========= Benchmark Result =========")

	fmt.Printf("Duration:        %v\n", duration)
	fmt.Printf("Total Requests:  %d\n", total)
	fmt.Printf("Success:         %d\n", m.success)
	fmt.Printf("Failed:          %d\n", m.fail)

	fmt.Printf("QPS:             %.2f\n", qps)

	fmt.Printf("Avg Latency:     %.2f ms\n", avg)
	fmt.Printf("P50 Latency:     %.2f ms\n", float64(p50)/1000)
	fmt.Printf("P90 Latency:     %.2f ms\n", float64(p90)/1000)
	fmt.Printf("P99 Latency:     %.2f ms\n", float64(p99)/1000)
}

func percentile(data []int64, p int) int64 {

	if len(data) == 0 {
		return 0
	}

	k := int(float64(len(data)) * float64(p) / 100.0)

	if k >= len(data) {
		k = len(data) - 1
	}

	return data[k]
}

```

# 两种方式总结:
在实际工程实践中，这两种性能测试方式往往是结合使用的。固定请求数量的压测更适合用于开发阶段的性能对比和优化验证，例如评估某个代码优化是否有效。由于每次测试处理的请求数量完全一致，因此更容易进行横向对比。而基于时间窗口的压测则更适合用于系统级性能评估，例如测试系统在持续负载下的吞吐能力、观察延迟分布情况，以及评估系统在高并发场景下的稳定性。

总体来看，**固定请求数量的压测更强调测试过程的可控性和结果的可重复性**，而**基于时间窗口的压测更强调负载的持续性和对真实场景的模拟**。在 RPC 框架的性能评估过程中，同时提供这两种测试方式，可以从不同角度全面分析系统性能，从而更准确地定位性能瓶颈并指导后续优化工作
